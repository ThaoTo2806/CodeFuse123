#Bước 3 — Chạy evaluation trên model vừa train
# Tạo một cell mới trong Kaggle và chạy toàn bộ code dưới đây.
# Code sẽ nạp adapter fixed_v2 ở chế độ 4-bit, giữ 512 token cuối của prompt khi inference, dự đoán với temperature=0 và tính các chỉ số đánh giá.

import os
import re
import json
import torch
from datasets import load_dataset
from transformers import (
    AutoTokenizer,
    AutoModelForCausalLM,
    BitsAndBytesConfig,
)
from peft import PeftModel
from sklearn.metrics import accuracy_score, f1_score, classification_report

# 1. Đường dẫn
adapter_path = "/kaggle/working/hqcm_llama2_7b_chcl_fixed_v2"
test_path = "/kaggle/working/hqcm_chcl/test.json"
output_path = "/kaggle/working/hqcm_chcl_predictions.json"

assert os.path.isfile(os.path.join(adapter_path, "adapter_config.json")), (
    f"Không tìm thấy adapter_config.json trong {adapter_path}. "
    "Hãy kiểm tra lại thư mục model đã train."
)
assert os.path.isfile(test_path), f"Không tìm thấy {test_path}"
assert torch.cuda.is_available(), "Kaggle chưa bật GPU!"

# 2. Đọc cấu hình adapter và nạp tokenizer
with open(os.path.join(adapter_path, "adapter_config.json"), encoding="utf-8") as f:
    adapter_config = json.load(f)

base_model = adapter_config["base_model_name_or_path"]
print("Base model:", base_model)
print("LoRA adapter:", adapter_path)

tokenizer = AutoTokenizer.from_pretrained(
    base_model,
    trust_remote_code=True
)
# Giữ phần cuối prompt, nơi có thể chứa chỉ dẫn phân loại
tokenizer.truncation_side = "left"

# 3. Nạp base model 4-bit rồi gắn LoRA adapter
bnb_config = BitsAndBytesConfig(
    load_in_4bit=True,
    bnb_4bit_quant_type="nf4",
    bnb_4bit_compute_dtype=torch.float16,
    bnb_4bit_use_double_quant=True,
)

base = AutoModelForCausalLM.from_pretrained(
    base_model,
    quantization_config=bnb_config,
    device_map="auto",
    torch_dtype=torch.float16,
    trust_remote_code=True,
)

model = PeftModel.from_pretrained(base, adapter_path)
model.eval()

if tokenizer.pad_token_id is None:
    tokenizer.pad_token = tokenizer.eos_token

# 4. Đọc test.json
dataset = load_dataset(
    "json",
    data_files={"test": test_path},
    split="test"
)
print("Số mẫu test:", len(dataset))
assert len(dataset) == 518, (
    f"Tập test có {len(dataset)} mẫu, khác dự kiến 518. "
    "Hãy kiểm tra lại dataset."
)

def clean_label(text):
    """Chuẩn hóa nhãn, ưu tiên nhãn hợp lệ xuất hiện trong output."""
    return str(text).strip().lower().strip("`'\".,:;!? ")

gold = [clean_label(row["answer"]) for row in dataset]
label_set = set(gold)
print("Các nhãn trong test:", sorted(label_set))

# 5. Sinh dự đoán
results = []
predictions = []
invalid = 0

for i, row in enumerate(dataset):
    prompt = row["prompt"]
    expected = gold[i]

    inputs = tokenizer(
        prompt,
        return_tensors="pt",
        truncation=True,
        max_length=512
    ).to("cuda")

    with torch.inference_mode():
        output = model.generate(
            **inputs,
            do_sample=False,       # tương đương temperature = 0
            max_new_tokens=16,
            num_beams=1,
            pad_token_id=tokenizer.pad_token_id,
            eos_token_id=tokenizer.eos_token_id,
        )

    new_tokens = output[0, inputs["input_ids"].shape[1]:]
    raw = tokenizer.decode(
        new_tokens,
        skip_special_tokens=True
    ).strip()

    # Thử lấy nhãn chính xác từ dòng đầu
    first_line = raw.splitlines()[0].strip() if raw else ""
    pred = clean_label(first_line)

    # Nếu model thêm lời giải thích, tìm nhãn hợp lệ trong output
    if pred not in label_set:
        words = re.findall(r"[A-Za-z][A-Za-z0-9_-]*", raw.lower())
        matches = [w for w in words if w in label_set]
        if matches:
            pred = matches[0]

    is_invalid = pred not in label_set
    if is_invalid:
        invalid += 1

    predictions.append(pred)
    results.append({
        "prompt": prompt,
        "expected_answer": row["answer"],
        "actual_answer": raw,
        "normalized_prediction": pred,
        "invalid": is_invalid,
    })

    if (i + 1) % 25 == 0 or i == 0 or i + 1 == len(dataset):
        print(
            f"[{i+1}/{len(dataset)}] "
            f"expected={expected!r} | predicted={pred!r} | raw={raw[:100]!r}"
        )

# 6. Lưu kết quả thô và nhãn đã chuẩn hóa
with open(output_path, "w", encoding="utf-8") as f:
    json.dump(results, f, ensure_ascii=False, indent=2)

# Macro/weighted F1 tính trên toàn bộ mẫu; dự đoán không hợp lệ vẫn bị tính sai
all_labels = sorted(label_set)
accuracy = accuracy_score(gold, predictions)
macro_f1 = f1_score(
    gold, predictions, labels=all_labels,
    average="macro", zero_division=0
)
weighted_f1 = f1_score(
    gold, predictions, labels=all_labels,
    average="weighted", zero_division=0
)

print("\n========== EVALUATION RESULTS ==========")
print(f"Test samples : {len(gold)}")
print(f"Accuracy     : {accuracy:.4f}")
print(f"Macro F1     : {macro_f1:.4f}")
print(f"Weighted F1  : {weighted_f1:.4f}")
print(f"Invalid      : {invalid}/{len(gold)} ({invalid/len(gold):.2%})")
print("\nClassification report:")
print(classification_report(
    gold, predictions,
    labels=all_labels,
    zero_division=0,
    digits=4
))
print("Predictions saved to:", output_path)
